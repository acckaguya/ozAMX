cmake_minimum_required(VERSION 3.16)
project(ozamx_8454h_v2 LANGUAGES C CXX ASM)

option(OZAMX_ENABLE_OPENMP "Enable OpenMP execution" ON)
option(OZAMX_BUILD_DIAGNOSTICS "Build native diagnostics" ON)
set(OZAMX_CACHE_BLOCK 128 CACHE STRING "Output cache block")
set(OZAMX_PREP_GRAIN 16384 CACHE STRING "Input elements per preprocessing worker")
set(OZAMX_MIN_COMPUTE_WORK 8 CACHE STRING "Small-shape output tasks per worker")
set(OZAMX_TILELOAD_HINT 1 CACHE STRING "AMX load hint: 0=TILELOADD, 1=TILELOADDT1")
set(OZAMX_LOAD_SCHEDULE 0 CACHE STRING "FP64 load schedule")
set(OZAMX_LOOP_ORDER 0 CACHE STRING "Compute loop order")
set(OZAMX_MN_ORDER 0 CACHE STRING "Output task order")
set(OZAMX_NUMA_REPLICATE_B 0 CACHE STRING "Replicate packed B by NUMA node for large teams")

set(OZAMX_TUNING_DEFINITIONS)
foreach(knob CACHE_BLOCK PREP_GRAIN MIN_COMPUTE_WORK TILELOAD_HINT LOAD_SCHEDULE LOOP_ORDER MN_ORDER NUMA_REPLICATE_B)
  if(NOT "${OZAMX_${knob}}" MATCHES "^[0-9]+$")
    message(FATAL_ERROR "OZAMX_${knob} must be a nonnegative integer")
  endif()
  list(APPEND OZAMX_TUNING_DEFINITIONS "OZAMX_${knob}=${OZAMX_${knob}}")
endforeach()

add_library(ozamx STATIC src/runtime.cpp src/gemm.cpp src/kernels.S)
target_include_directories(ozamx PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ozamx PUBLIC cxx_std_17)
target_compile_definitions(ozamx PRIVATE ${OZAMX_TUNING_DEFINITIONS})
target_compile_options(ozamx PRIVATE -Wall -Wextra -Wpedantic -mamx-tile -mamx-bf16
  $<$<CONFIG:Release>:-O3>)
set_source_files_properties(src/gemm.cpp PROPERTIES COMPILE_OPTIONS "-march=sapphirerapids;-ffp-contract=off")
if(OZAMX_ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  target_link_libraries(ozamx PUBLIC OpenMP::OpenMP_CXX)
endif()

add_executable(ozamx_cli src/main.cpp)
target_link_libraries(ozamx_cli PRIVATE ozamx)

include(CTest)
if(BUILD_TESTING)
  add_executable(ozamx_validation tests/validation.cpp)
  target_link_libraries(ozamx_validation PRIVATE ozamx)
  target_compile_options(ozamx_validation PRIVATE -mamx-tile -mamx-bf16)
  add_test(NAME ozamx_validation COMMAND ozamx_validation)
  add_executable(ozamx_late_path_probe tests/final_late_path_probe.cpp)
  target_link_libraries(ozamx_late_path_probe PRIVATE ozamx)
  target_compile_options(ozamx_late_path_probe PRIVATE -march=sapphirerapids -ffp-contract=off)
  add_test(NAME ozamx_late_path_probe COMMAND ozamx_late_path_probe)
  add_executable(ozamx_alias_output_probe tests/bounded_commit_alias_probe.cpp)
  target_link_libraries(ozamx_alias_output_probe PRIVATE ozamx)
  target_compile_options(ozamx_alias_output_probe PRIVATE -march=sapphirerapids -ffp-contract=off)
  add_test(NAME ozamx_alias_output_probe COMMAND ozamx_alias_output_probe)
endif()

if(OZAMX_BUILD_DIAGNOSTICS)
  add_executable(ozamx_native_kernel_audit tools/native_kernel_audit.cpp src/runtime.cpp src/kernels.S)
  target_include_directories(ozamx_native_kernel_audit PRIVATE include)
  target_compile_features(ozamx_native_kernel_audit PRIVATE cxx_std_17)
  target_compile_definitions(ozamx_native_kernel_audit PRIVATE ${OZAMX_TUNING_DEFINITIONS})
  target_compile_options(ozamx_native_kernel_audit PRIVATE -O3 -march=sapphirerapids -ffp-contract=off)
  if(OZAMX_ENABLE_OPENMP)
    target_link_libraries(ozamx_native_kernel_audit PRIVATE OpenMP::OpenMP_CXX)
  endif()
  add_executable(ozamx_platform_baseline tools/platform_baseline.cpp)
  target_compile_features(ozamx_platform_baseline PRIVATE cxx_std_17)
  target_compile_options(ozamx_platform_baseline PRIVATE -O3 -march=sapphirerapids -ffp-contract=off)
  if(OZAMX_ENABLE_OPENMP)
    target_link_libraries(ozamx_platform_baseline PRIVATE OpenMP::OpenMP_CXX)
  endif()
  add_executable(ozamx_optimization_benchmark bench/optimization_benchmark.cpp)
  target_link_libraries(ozamx_optimization_benchmark PRIVATE ozamx)
endif()
