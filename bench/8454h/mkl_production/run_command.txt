# Final production-linked build and full matrix run. Build and measurement both
# hold the shared perf lock; this links the frozen fresh CMake library without
# rebuilding or changing its production sources.
flock -x /tmp/ozamx-perf.lock bash -c '
  mkdir -p /tmp/ozamx-mkl-production-final
  g++ -O3 -std=c++17 -Wall -Wextra -Wpedantic -mamx-tile -mamx-bf16 -fopenmp \
    -I/home/caiweili/ozAMX/include -I/home/caiweili/ozAMX/third_party/code/mkl-current/include \
    /home/caiweili/ozAMX/bench/8454h/mkl_production/benchmark_production.cpp \
    /tmp/ozamx-production-final/libozamx.a \
    -L/home/caiweili/ozAMX/third_party/code/mkl-current/lib \
    -Wl,-rpath,/home/caiweili/ozAMX/third_party/code/mkl-current/lib \
    -Wl,--no-as-needed -l:libmkl_rt.so.3 -ldl \
    -o /tmp/ozamx-mkl-production-final/ozamx_benchmark && \
  LD_LIBRARY_PATH=/home/caiweili/ozAMX/third_party/code/mkl-current/lib \
  OMP_NUM_THREADS=64 OMP_PLACES=cores OMP_PROC_BIND=close OMP_DYNAMIC=FALSE \
  MKL_NUM_THREADS=64 MKL_DYNAMIC=FALSE MKL_THREADING_LAYER=GNU MKL_INTERFACE_LAYER=LP64 \
  OZAMX_WORKSPACE_CACHE_BYTES=2147483648 taskset -c 0-63 \
    /tmp/ozamx-mkl-production-final/ozamx_benchmark \
    --sizes 256,1024,4096,8192 --threads 64 --reps 5 --precision all \
    --budgets 6,10,15,21 --seed 260904 --distribution uniform \
    --checkpoint /home/caiweili/ozAMX/bench/8454h/mkl_production/checkpoint_final_prod \
    --output /home/caiweili/ozAMX/bench/8454h/mkl_production/full_production.csv
'
